#include <iostream>
using namespace std;

// Node class
class node {
public:
    int data;
    node* next;

    // Constructor
    node(int x) {
        data = x;
        next = NULL;
    }
};

// Queue class using Linked List
class queue {
private:
    node* front;
    node* rear;

public:
    // Constructor
    queue() {
        front = rear = NULL;
    }

    // Check whether queue is empty
    int empty(void) {
        if (front == NULL) {
            return 1;
        }
        else {
            return 0;
        }
    }

    // Return front element
    int peek(void) {
        if (empty()) {
            cout << "\nQueue is empty";
            return -1;
        }

        return front->data;
    }

    // Insert element at rear
    void enqueue(int x) {
        node* p = new node(x);

        if (front == NULL) {
            front = p;
            rear = p;
        }
        else {
            rear->next = p;
            rear = p;
        }
    }

    // Remove element from front
    int dequeue(void) {
        if (empty()) {
            cout << "\nQueue is empty";
            return -1;
        }

        int x = front->data;
        node* p = front;

        front = front->next;

        if (front == NULL) {
            rear = NULL;
        }

        delete p;
        return x;
    }

    // Display queue elements
    void display(void) {
        node* q = front;

        cout << "\nQueue elements :";

        while (q != NULL) {
            cout << "\t" << q->data;
            q = q->next;
        }
    }
};

// Main function
int main() {
    int x;
    queue q;

    // Insert elements into queue
    q.enqueue(27);
    q.enqueue(34);
    q.enqueue(42);
    q.enqueue(58);
    q.enqueue(65);
    q.enqueue(70);

    // Display queue
    q.display();

    // Delete front element
    if (!q.empty()) {
        x = q.dequeue();
        cout << "\nRemoved: " << x;
    }

    // Display queue after deletion
    q.display();

    // Display front element
    if (!q.empty()) {
        x = q.peek();
        cout << "\nFront element is: " << x;
    }

    return 0;
}
