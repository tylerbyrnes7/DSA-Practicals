#include<iostream>
using namespace std;

// Function for Bubble Sort
void bubble_sort(int a[], int n)
{
    for(int i = 0; i < n - 1; i++)
    { 
        for(int j = 0; j < n - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
}

int main()
{
    int n;
    int a[10];

    // Accept array size
    cout << "Enter the size of the array: ";
    cin >> n;

    // Validate array size
    if(n <= 0 || n > 10)
    {
        cout << "Invalid array size!" << endl;
        return 0;
    }

    // Accept array elements
    cout << "Enter " << n << " integers in any order:\n";

    for(int i = 0; i < n; i++)
    {
        cin >> a[i];
    }

    // Call Bubble Sort function
    bubble_sort(a, n);

    // Display sorted array
    cout << "Sorted array: ";

    for(int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    cout << endl;

    return 0;
}
